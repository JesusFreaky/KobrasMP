# Event-driven Cawa base reconciliation. No polling.
on_startup = {
	kob_reconcile_cawa_base_effect = yes
}

on_reform_enacted = {
	kob_reconcile_cawa_base_effect = yes
}

on_reform_changed = {
	kob_reconcile_cawa_base_effect = yes
}

# THIS = Province; owner = recruiting country.
on_cawa_raised = {
	owner = {
		if = {
			limit = { tag = PAP }
			add_manpower = -0.5
			add_treasury = -15
		}
	}
}
